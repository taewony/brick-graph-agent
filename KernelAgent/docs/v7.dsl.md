  <milestones>
    
    <!-- Stage 1: GEMM (Matrix Multiplication) -->
    <stage id="1_GEMM" complexity="Low">
      <bottleneck>Shared memory bank conflicts and register occupancy limits.</bottleneck>
      <solution>Implement tiled GEMM with cuTile, loading sub-matrices into shared memory.</solution>
      <code_primitive lang="python">
        # cuTile Tiled MatMul Kernel Example
        @ct.kernel(occupancy=2)
        def gemm_kernel(A, B, C, TILE_M: ConstInt, TILE_N: ConstInt, TILE_K: ConstInt):
            tx = ct.threadIdx(0)
            ty = ct.threadIdx(1)
            # Load blocks to Shared Memory and compute MMA
            a_tile = ct.load(A, index=(ct.bid(0) * TILE_M, 0))
            b_tile = ct.load(B, index=(0, ct.bid(1) * TILE_N))
            acc = ct.mma(a_tile, b_tile)
            ct.store(C, index=(ct.bid(0) * TILE_M, ct.bid(1) * TILE_N), tile=acc)
      </code_primitive>
    </stage>

    <!-- Stage 2: FMHA (Fused Multi-Head Attention) -->
    <stage id="2_FMHA" complexity="Medium">
      <bottleneck>Memory-bandwidth limit due to intermediate attention matrix write-back.</bottleneck>
      <solution>Fuse attention computation by storing intermediate scores in register tiles; apply causal mask and scale online.</solution>
      <code_primitive lang="python">
        # Online softmax and causal masking inside cuTile prefill loop
        for j in range(0, Tc):
            k = ct.load(K, index=(batch_idx, off_kv_h, 0, j), shape=(1, 1, TILE_D, TILE_N)).reshape((TILE_D, TILE_N))
            qk = ct.full((TILE_M, TILE_N), 0., dtype=np.float32)
            qk = ct.mma(q, k, qk)
            if CAUSAL and j >= mask_start:
                offs_n = j * TILE_N + offs_n_tile
                mask = ct.where(offs_m >= offs_n, 0.0, -np.inf)
                qk += mask
            # Online softmax reduction
            m_ij = max(m_i, ct.max(qk, axis=-1, keepdims=True) * qk_scale)
            qk = qk * qk_scale - m_ij
            p = ct.exp2(qk, flush_to_zero=True)
            l_ij = ct.sum(p, axis=-1, keepdims=True)
            alpha = ct.exp2(m_i - m_ij, flush_to_zero=True)
            l_i = l_i * alpha + l_ij
            acc = acc * alpha
      </code_primitive>
    </stage>

    <!-- Stage 3: Single User LLM Inference -->
    <stage id="3_Single_User_LLM" complexity="High">
      <bottleneck>High Python runtime launch overhead (ct.launch ctypes costs) in token-by-token decoding.</bottleneck>
      <solution>Record and replay the decode execution path using CUDA Graphs with static pre-allocated memory buffers.</solution>
      <code_primitive lang="python">
        # CUDA Graph Capture and Replay for Decode Loop
        static_input = torch.zeros(batch_size, dtype=torch.int64, device='cuda')
        static_kv_cache = torch.zeros(layers, batch_size, seq_len, head_dim, device='cuda')
        static_logits = torch.zeros(batch_size, vocab_size, device='cuda')

        # Warmup execution
        for _ in range(3):
            outputs = model(static_input, static_kv_cache)
        
        # Capture into Graph
        g = torch.cuda.CUDAGraph()
        with torch.cuda.graph(g):
            static_logits.copy_(model(static_input, static_kv_cache))
        
        # Generation loop replay
        for step in range(max_new_tokens):
            static_input.copy_(next_token_tensor)
            g.replay()
            next_token_tensor.copy_(sample(static_logits))
      </code_primitive>
    </stage>
